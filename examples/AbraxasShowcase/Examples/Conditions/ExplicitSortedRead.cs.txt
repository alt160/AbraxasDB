{
    using var reader = customers.Readers.OpenSorted(".Name", descending: true);
    int shown = 0;
    while (shown < 8 && reader.Read())
    {
        Console.WriteLine($"{reader.ID}: {reader.Get().Name}");
        shown++;
    }
}
