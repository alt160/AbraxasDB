// Name has no pattern index in the group setup; this condition searches stored names.
// The group context above owns the open RecordBase and stores.
{
    var condition = customers.Where.Prop(customer => customer.Name).MatchesPattern("1$", false).EndCondition;
    foreach (ulong recordID in customers.IterateRecordIDs(condition).Take(8))
    {
        ShowcaseCustomer customer = customers.Get(recordID);
        Console.WriteLine($"{recordID}: {customer.Name}");
    }
}
