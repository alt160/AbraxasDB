// The type codec handles every ShowcaseHookToken in TypeTokens.
// documents belongs to the group context and remains available until that context is disposed.
{
    using var reader = documents.Readers.Open();
    if (reader.Read())
    {
        ShowcaseHookDocument document = documents.Get(reader.ID);
        Console.WriteLine(string.Join(", ", document.TypeTokens.Select(token => token.Code)));
    }
}
